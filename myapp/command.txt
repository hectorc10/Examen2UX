controller/Notes.js:
const { Note, Author } = require("../models");



let getNotes = async (req, res) => {

  try {

    let notes = await Note.findAll({

      order: [['priority', 'DESC']]

    });



    if (notes.length <= 0) {

      return res.status(204).json({ status: 204, message: "No hay notas" });

    }



    res.status(200).json({ status: 200, data: notes });



  } catch (error) {

    res.status(500).json({ status: 500, message: error.message });

  }

};



let createNote = async (req, res) => {

  try {

    const { title, content, priority, authorId } = req.body;



    const author = await Author.findByPk(authorId);

    if (!author) {

      return res.status(404).json({ status: 404, message: "Autor no encontrado" });

    }



    const newNote = await Note.create({ title, content, priority, authorId });



    res.status(201).json({ status: 201, data: newNote });



  } catch (error) {

    res.status(500).json({ status: 500, message: error.message });

  }

};



module.exports = { getNotes, createNote };
